Fterug := Fz * sin(hoek) ' volgt uit constructie
Fres := Fterug - Fwr                ' resulterende kracht
a := Fres / m                  ' 2e wet van Newton
v := v + a * dt                ' baansnelheid neemt toe (start: negatief!)
ds := v * dt                     ' in dt afgelegde afstand (over cirkelbaan)
vorighoek = hoek
hoek := hoek + ds * f    'van afstand (over cirkelbaan) naar hoek (in graden)

Fwr := 0.5 * rho * opp * cw * (v*v)    'formule voor luchtwrijving
Fwr := sign(v) * Fwr                   'het teken (plus/min) overnemen (met functie sign()); richting van wrijvingskracht!!

t := t + dt                      ' tijd bijhouden

if hoek > vorighoek then
  halfper = 1
endif

if hoek < vorighoek then
  if halfper = 1 then
    T = t / run
    run = run + 1
    halfper = 0
  endif
endif
